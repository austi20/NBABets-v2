# Per-Market ECE Thresholds (Dynamic Null-Floor Gate) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace hardcoded ECE thresholds with a dynamic, self-calibrating gate that compares each market's observed ECE to the null floor a perfectly-calibrated model would produce at that market's sample size, and capture OOF once to drive both this and the later fold sweep.

**Architecture:** A new pure module `app/training/calibration_ece.py` holds the ECE metric, the bootstrap null distribution, the per-market gate, and OOF dump/load io. The calibration pass in `pipeline.py` calls the gate (deleting the magic-number map) and stores floors in `calibration_diagnostics`. A flag-gated dump persists per-fold OOF. An offline script turns the dump into a committed findings report. `selective_retrain` reads the persisted per-market floor.

**Tech Stack:** Python 3.12, NumPy, pytest. No new dependencies.

**Spec:** `docs/plans/specs/calibration-ece-thresholds-spec.md`

```python
# app/training/calibration_ece.py
from __future__ import annotations

import numpy as np

DEFAULT_N_BINS = 10


def compute_ece(probs: np.ndarray, labels: np.ndarray, n_bins: int = DEFAULT_N_BINS) -> float:
    """Count-weighted |confidence - accuracy| over equal-width bins on [0, 1].

    Extracted verbatim from the prior inline pipeline computation so production and
    analysis share one implementation.
    """
    probs = np.asarray(probs, dtype=float)
    labels = np.asarray(labels, dtype=float)
    if probs.size == 0:
        return 0.0
    bin_edges = np.linspace(0.0, 1.0, n_bins + 1)
    bin_indices = np.digitize(probs, bin_edges, right=True).clip(1, n_bins) - 1
    ece = 0.0
    for b in range(n_bins):
        mask = bin_indices == b
        if not mask.any():
            continue
        bin_conf = float(probs[mask].mean())
        bin_acc = float(labels[mask].mean())
        ece += mask.sum() * abs(bin_conf - bin_acc)
    return float(ece / max(probs.size, 1))
```

```python
# append to app/training/calibration_ece.py

def null_ece_distribution(
    probs: np.ndarray,
    *,
    n_bins: int = DEFAULT_N_BINS,
    n_boot: int = 2000,
    rng: np.random.Generator,
) -> np.ndarray:
    """ECE values a PERFECTLY calibrated model would produce at this probability
    vector. Each draw treats ``probs`` as ground truth: y_i ~ Bernoulli(probs_i)."""
    probs = np.asarray(probs, dtype=float)
    out = np.empty(int(n_boot), dtype=float)
    for i in range(int(n_boot)):
        labels = (rng.random(probs.size) < probs).astype(float)
        out[i] = compute_ece(probs, labels, n_bins=n_bins)
    return out


def null_floor(
    probs: np.ndarray,
    *,
    percentile: float = 99.0,
    n_bins: int = DEFAULT_N_BINS,
    n_boot: int = 2000,
    rng: np.random.Generator,
) -> float:
    """percentile-th percentile of the null ECE distribution: the max ECE consistent
    with perfect calibration at this sample size and probability shape."""
    dist = null_ece_distribution(probs, n_bins=n_bins, n_boot=n_boot, rng=rng)
    return float(np.percentile(dist, percentile))
```

```python
# append to app/training/calibration_ece.py

def market_ece_gate(
    probs: np.ndarray,
    labels: np.ndarray,
    *,
    percentile: float = 99.0,
    n_bins: int = DEFAULT_N_BINS,
    n_boot: int = 2000,
    rng: np.random.Generator,
) -> dict[str, float | int | bool]:
    """Self-calibrating gate: compare observed ECE to the null floor at this size.

    Returns ece, null_floor (at ``percentile``), null_floor_p95, passed, sample_count.
    One bootstrap is reused for both percentiles.
    """
    probs = np.asarray(probs, dtype=float)
    labels = np.asarray(labels, dtype=float)
    dist = null_ece_distribution(probs, n_bins=n_bins, n_boot=n_boot, rng=rng)
    ece = compute_ece(probs, labels, n_bins=n_bins)
    floor = float(np.percentile(dist, percentile))
    floor95 = float(np.percentile(dist, 95.0))
    return {
        "ece": round(ece, 6),
        "null_floor": round(floor, 6),
        "null_floor_p95": round(floor95, 6),
        "passed": bool(ece <= floor),
        "sample_count": int(probs.size),
    }
```

```python
# append to app/training/calibration_ece.py
import json
from pathlib import Path


def dump_calibration_oof(
    oof_raw: dict[str, list[np.ndarray]],
    oof_labels: dict[str, list[np.ndarray]],
    path: str | Path,
    *,
    timing: dict | None = None,
) -> None:
    """Persist per-fold OOF probs/labels per market (preserves fold granularity for
    the Phase 2 fold sweep). Arrays are written as plain lists."""
    markets: dict[str, dict] = {}
    for market_key, fold_arrays in oof_raw.items():
        label_arrays = oof_labels.get(market_key, [])
        markets[market_key] = {
            "fold_probs": [np.asarray(a, dtype=float).tolist() for a in fold_arrays],
            "fold_labels": [np.asarray(a, dtype=float).tolist() for a in label_arrays],
        }
    payload = {"markets": markets, "timing": timing or {}}
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload), encoding="utf-8")


def load_calibration_oof(path: str | Path) -> dict:
    return json.loads(Path(path).read_text(encoding="utf-8"))
```

```python
# replace build_retrain_market_list in app/services/brain/actions/selective_retrain.py
def build_retrain_market_list(
    signals: list[dict[str, Any]],
    ece_threshold: float = 0.10,
    market_floors: dict[str, float] | None = None,
) -> list[str]:
    """Markets needing retraining: ECE above the market's persisted null floor when
    available, else the ``ece_threshold`` fallback."""
    floors = market_floors or {}
    markets: list[str] = []
    for sig in signals:
        market = sig.get("market")
        ece = sig.get("ece", 0.0)
        threshold = float(floors.get(market, ece_threshold)) if market else ece_threshold
        if market and ece > threshold:
            markets.append(market)
    return sorted(set(markets))
```

```python
def summarize_fold_subsets(
    fold_probs: list[np.ndarray],
    fold_labels: list[np.ndarray],
    *,
    fold_counts: list[int],
    percentile: float = 99.0,
    n_bins: int = DEFAULT_N_BINS,
    n_boot: int = 2000,
    rng: np.random.Generator,
) -> list[dict]:
    """For each candidate fold count k, take the first k folds, concatenate their OOF,
    and report the gate result. Drives the floor(market, fold_count) table."""
    rows: list[dict] = []
    total = len(fold_probs)
    for k in fold_counts:
        k = min(k, total)
        probs = np.concatenate([np.asarray(p, dtype=float) for p in fold_probs[:k]])
        labels = np.concatenate([np.asarray(y, dtype=float) for y in fold_labels[:k]])
        gate = market_ece_gate(
            probs, labels, percentile=percentile, n_bins=n_bins, n_boot=n_boot, rng=rng
        )
        rows.append({"fold_count": k, **gate})
    return rows
```

```python
# scripts/analyze_calibration_ece.py
"""Offline analysis of a captured calibration OOF dump.

Usage:
    .venv/Scripts/python.exe scripts/analyze_calibration_ece.py \
        --dump data/artifacts/calibration_oof_dump.json \
        --out docs/plans/reports/calibration-ece-findings.md
"""
from __future__ import annotations

import argparse
from pathlib import Path

import numpy as np

from app.training.calibration_ece import load_calibration_oof, summarize_fold_subsets

FOLD_COUNTS = [2, 4, 6, 8, 10, 12, 15, 20]


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--dump", default="data/artifacts/calibration_oof_dump.json")
    parser.add_argument("--out", default="docs/plans/reports/calibration-ece-findings.md")
    parser.add_argument("--percentile", type=float, default=99.0)
    parser.add_argument("--n-boot", type=int, default=2000)
    args = parser.parse_args()

    data = load_calibration_oof(args.dump)
    rng = np.random.default_rng(42)
    lines: list[str] = ["# Calibration ECE Findings", ""]
    lines.append(f"Source: `{args.dump}` | null percentile: p{args.percentile:.0f} | n_boot: {args.n_boot}")
    lines.append("")
    for market, payload in data["markets"].items():
        fold_probs = [np.asarray(p, dtype=float) for p in payload["fold_probs"]]
        fold_labels = [np.asarray(y, dtype=float) for y in payload["fold_labels"]]
        rows = summarize_fold_subsets(
            fold_probs, fold_labels, fold_counts=FOLD_COUNTS,
            percentile=args.percentile, n_boot=args.n_boot, rng=rng,
        )
        lines.append(f"## {market}")
        lines.append("")
        lines.append("| folds | n | ECE | null_floor_p99 | null_floor_p95 | verdict |")
        lines.append("|---|---|---|---|---|---|")
        for r in rows:
            verdict = "OK" if r["passed"] else "MISCALIBRATED"
            lines.append(
                f"| {r['fold_count']} | {r['sample_count']} | {r['ece']:.4f} | "
                f"{r['null_floor']:.4f} | {r['null_floor_p95']:.4f} | {verdict} |"
            )
        lines.append("")
    out = Path(args.out)
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text("\n".join(lines), encoding="utf-8")
    print(f"wrote {out}")


if __name__ == "__main__":
    main()
```
